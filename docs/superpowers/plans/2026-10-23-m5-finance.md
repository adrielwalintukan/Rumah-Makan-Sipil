# Catatan Keuangan: Pengeluaran + Buku Kas + Laba Kotor Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Mengimplementasikan fitur catatan pengeluaran operasional manual, agregasi buku kas harian/bulanan, dan ringkasan laba kotor di `/admin/keuangan` sesuai PRD §4.2 (D9), §5 (US-15), dan architecture.md §4 & §5.3.

**Architecture:** Fungsi SQL PostgreSQL `public.cashbook(p_from_date, p_to_date)` berkeamanan `security definer` dan guard `public.is_owner()` menghitung total pemasukan pesanan lunas non-batal, total pengeluaran per kategori, laba kotor (`income - expense`), serta rincian harian `dailyBreakdown`. Modul server helper `lib/admin-finance.ts` menyediakan validasi Zod untuk query periode dan payload pengeluaran, serta helper `getFinanceData`. Endpoint REST API memfasilitasi operasi data buku kas dan CRUD pengeluaran. Antarmuka `/admin/keuangan` dibangun dengan pola Server Component (PPR & Suspense) dan komponen mobile-first teruji aksesibilitas WCAG AA dan target sentuh minimal 44x44 px.

**Tech Stack:** Next.js 16 (App Router), React 19, TypeScript strict, Tailwind CSS, Supabase (PostgreSQL 15, RLS, RPC), Vitest, Playwright.

**Spec:** `docs/superpowers/specs/2026-10-23-m5-finance-design.md`

## Global Constraints

- Nol karakter em dash (`—`) di seluruh UI copy, pesan error, nama file, komentar kode, dan log (antislop R-02).
- Desain mobile-first responsif: diuji pada viewport 360 px tanpa scroll horizontal, layout mengalir rapi.
- Semua target sentuh interaktif (tombol, tab, input, link aksi) berukuran minimal 44x44 px (`min-h-[44px] min-w-[44px]`).
- Kontras teks memenuhi standar WCAG AA (>= 4.5:1 untuk teks normal; `text-ink-900`, `text-ink-700`, `text-ink-600` di atas latar terang; indikator surplus/defisit semantik).
- Nol dependensi eksternal baru: gunakan utilitas bawaan repo (`lib/format.ts`, `@/components/ui/*`, Lucide icons, Tailwind).
- Semua nilai nominal uang disimpan dan diproses dalam integer Rupiah (tanpa desimal).
- Semua pengelompokan hari/tanggal bertransaksi memakai zona waktu WITA (`Asia/Makassar`).
- Di Next.js 16 App Router, `searchParams` pada Server Component Page wajib di-await (`const params = await searchParams;`).
- Commit messages menggunakan Conventional Commits dalam Bahasa Indonesia tanpa footer atau attribution AI (aturan repo AGENTS.md).
- Shell agent dilarang menjalankan migrasi langsung ke Supabase hosted; semua migrasi hosted dipandu untuk dijalankan di terminal pengguna (`npx supabase db push`).

-- Migrasi: Fungsi SQL cashbook untuk buku kas dan laba kotor (Milestone M5: Issue #22)

create or replace function public.cashbook(
  p_from_date date,
  p_to_date date
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_income bigint;
  v_expense bigint;
  v_profit bigint;
  v_expense_bahan bigint;
  v_expense_operasional bigint;
  v_expense_lainnya bigint;
  v_daily jsonb;
begin
  -- 1. Penegakan Keamanan: Hanya pemilik yang berhak
  if not public.is_owner() then
    raise exception 'FORBIDDEN: Hanya akun pemilik yang memiliki hak akses' using errcode = '42501';
  end if;

  -- 2. Validasi parameter tanggal
  if p_from_date is null or p_to_date is null then
    raise exception 'INVALID_PARAMETER: Tanggal awal dan akhir harus diisi' using errcode = '22023';
  end if;

  if p_from_date > p_to_date then
    raise exception 'INVALID_PARAMETER: Tanggal awal tidak boleh lebih besar dari tanggal akhir' using errcode = '22023';
  end if;

  -- 3. Hitung total pemasukan dari pesanan lunas non-batal
  select coalesce(sum(o.total), 0)
    into v_income
    from public.orders o
   where o.order_date between p_from_date and p_to_date
     and o.payment_status = 'lunas'
     and o.status <> 'batal';

  -- 4. Hitung total pengeluaran dan rincian per kategori
  select coalesce(sum(e.amount), 0),
         coalesce(sum(case when e.category = 'bahan' then e.amount else 0 end), 0),
         coalesce(sum(case when e.category = 'operasional' then e.amount else 0 end), 0),
         coalesce(sum(case when e.category = 'lainnya' then e.amount else 0 end), 0)
    into v_expense,
         v_expense_bahan,
         v_expense_operasional,
         v_expense_lainnya
    from public.expenses e
   where e.expense_date between p_from_date and p_to_date;

  -- 5. Hitung laba kotor total
  v_profit := v_income - v_expense;

  -- 6. Agregasi rincian buku kas harian
  with dates as (
    select distinct o.order_date as dt
      from public.orders o
     where o.order_date between p_from_date and p_to_date
       and o.payment_status = 'lunas'
       and o.status <> 'batal'
    union
    select distinct e.expense_date as dt
      from public.expenses e
     where e.expense_date between p_from_date and p_to_date
  ),
  daily_income as (
    select o.order_date as dt,
           coalesce(sum(o.total), 0) as income
      from public.orders o
     where o.order_date between p_from_date and p_to_date
       and o.payment_status = 'lunas'
       and o.status <> 'batal'
     group by o.order_date
  ),
  daily_expense as (
    select e.expense_date as dt,
           coalesce(sum(e.amount), 0) as expense
      from public.expenses e
     where e.expense_date between p_from_date and p_to_date
     group by e.expense_date
  )
  select coalesce(
           jsonb_agg(
             jsonb_build_object(
               'date', to_char(d.dt, 'YYYY-MM-DD'),
               'income', coalesce(di.income, 0),
               'expense', coalesce(de.expense, 0),
               'grossProfit', coalesce(di.income, 0) - coalesce(de.expense, 0)
             )
             order by d.dt desc
           ),
           '[]'::jsonb
         )
    into v_daily
    from dates d
    left join daily_income di on di.dt = d.dt
    left join daily_expense de on de.dt = d.dt;

  -- 7. Bentuk hasil akhir jsonb
  return jsonb_build_object(
    'period', jsonb_build_object(
      'from', to_char(p_from_date, 'YYYY-MM-DD'),
      'to', to_char(p_to_date, 'YYYY-MM-DD')
    ),
    'summary', jsonb_build_object(
      'totalIncome', v_income,
      'totalExpense', v_expense,
      'grossProfit', v_profit,
      'expenseByCategory', jsonb_build_object(
        'bahan', v_expense_bahan,
        'operasional', v_expense_operasional,
        'lainnya', v_expense_lainnya
      )
    ),
    'dailyBreakdown', v_daily
  );
end;
$$;

revoke all on function public.cashbook(date, date) from public, anon;
grant execute on function public.cashbook(date, date) to authenticated;
```

    ),

    .string()
    .trim()
    .min(1, "Keterangan pengeluaran wajib diisi.")
    .max(100, "Keterangan pengeluaran maksimal 100 karakter."),
  amount: z.coerce
    .number()
    .int("Nominal pengeluaran harus berupa bilangan bulat Rupiah.")
    .positive("Nominal pengeluaran harus lebih besar dari 0."),

const calendarDateSchema = z
  .string()
  .regex(/^\d{4}-\d{2}-\d{2}$/, "Format tanggal harus YYYY-MM-DD.")
  .refine(isValidCalendarDate, "Tanggal kalender tidak valid.");

export function createFinanceQuerySchema(now: Date = new Date()) {
  const todayStr = getWitaTodayDateString(now);

  return z
    .object({
      preset: financePresetEnum.default("today"),
      from: calendarDateSchema.optional(),
      to: calendarDateSchema.optional(),
    })
    .refine(
      (data) => !data.from || data.from <= todayStr,
      {
        message: "Tanggal awal (from) tidak boleh melebihi hari ini (WITA).",
        path: ["from"],
      }
    )
    .refine(
      (data) => !data.to || data.to <= todayStr,
      {
        message: "Tanggal akhir (to) tidak boleh melebihi hari ini (WITA).",
        path: ["to"],
      }
    )
    .refine(
      (data) => {
        if (data.from && data.to) return data.from <= data.to;
        return true;
      },
      {
        message: "Tanggal awal (from) tidak boleh lebih besar dari tanggal akhir (to).",
        path: ["from"],
      }
    );
}

// 3. Tipe Data Kontrak

export interface Expense {
  id: string;
  expense_date: string;
  description: string;
  amount: number;
  category: ExpenseCategory;
  created_at: string;
}

export interface CashbookExpenseByCategory {
  bahan: number;
  operasional: number;
  lainnya: number;
}

export interface CashbookSummary {
  totalIncome: number;
  totalExpense: number;
  grossProfit: number;
  expenseByCategory: CashbookExpenseByCategory;
}

export interface DailyCashbookItem {
  date: string;
  income: number;
  expense: number;
  grossProfit: number;
}

export interface CashbookResult {
  period: { from: string; to: string };
  summary: CashbookSummary;
  dailyBreakdown: DailyCashbookItem[];
}

export interface FinanceData {
  cashbook: CashbookResult;
  expenses: Expense[];
}

// ------------------------------------------------------------
// 4. Perhitungan Rentang Tanggal
// ------------------------------------------------------------
export function getFinanceDateRange(
  preset: FinancePreset,
  customFrom?: string,
  customTo?: string,
  now: Date = new Date()
): { from: string; to: string } {
  const todayStr = getWitaTodayDateString(now);

  switch (preset) {
    case "today":
      return { from: todayStr, to: todayStr };
    case "this_month":
      return getWitaMonthRange(now);
    case "custom": {
      if (customFrom && customTo) {
        let from = customFrom;
        let to = customTo;
        if (from > to) {
          const temp = from;
          from = to;
          to = temp;
        }
        if (from > todayStr) from = todayStr;
        if (to > todayStr) to = todayStr;
        return { from, to };
      }
      return { from: todayStr, to: todayStr };
    }
    default:
      return { from: todayStr, to: todayStr };
  }
}

// 5. Pengambilan Data Lengkap Keuangan (Server Helper)
// ------------------------------------------------------------
export async function getFinanceData(
  supabase: SupabaseClient,
  query: FinanceQuery,
  now: Date = new Date()
): Promise<FinanceData> {
  const { from, to } = getFinanceDateRange(query.preset, query.from, query.to, now);

  const fallbackCashbook: CashbookResult = {
    period: { from, to },
    summary: {
      totalIncome: 0,
      totalExpense: 0,
      grossProfit: 0,
      expenseByCategory: { bahan: 0, operasional: 0, lainnya: 0 },
    },
    dailyBreakdown: [],
  };

  try {
    const [cashbookRes, expensesRes] = await Promise.all([
      supabase.rpc("cashbook", { p_from_date: from, p_to_date: to }),
      supabase
        .from("expenses")
        .select("*")
        .gte("expense_date", from)
        .lte("expense_date", to)
        .order("expense_date", { ascending: false })
        .order("created_at", { ascending: false }),
    ]);

    let cashbook = fallbackCashbook;
    if (cashbookRes.data && !cashbookRes.error) {
      cashbook = cashbookRes.data as CashbookResult;
    } else if (cashbookRes.error) {
      console.error("Gagal memanggil RPC cashbook:", cashbookRes.error);
    }

    let expenses: Expense[] = [];
    if (expensesRes.data && !expensesRes.error) {
      expenses = expensesRes.data as Expense[];
    } else if (expensesRes.error) {
      console.error("Gagal mengambil data expenses:", expensesRes.error);
    }

    return { cashbook, expenses };
  } catch (err) {
    console.error("Kesalahan tak terduga saat mengambil data keuangan:", err);
    return { cashbook: fallbackCashbook, expenses: [] };
  }
}

    const admin = await getAdminUser();
    if (!admin) {
      return NextResponse.json(
        {
          error: "UNAUTHORIZED",
          message: "Akses ditolak: sesi admin diperlukan.",
        },
        { status: 401 }
      );
    }

    const { searchParams } = new URL(request.url);
    const rawParams: Record<string, string> = {};
    searchParams.forEach((val, key) => {
      if (val.trim() !== "") {
        rawParams[key] = val;
      }
    });

    const parsed = financeQuerySchema.safeParse(rawParams);
    if (!parsed.success) {
      return NextResponse.json(
        {
          error: "BAD_REQUEST",
          message: "Parameter query tidak valid.",
          details: parsed.error.issues,
        },
        { status: 400 }
      );
    }

    const { from, to } = getFinanceDateRange(
      parsed.data.preset,
      parsed.data.from,
      parsed.data.to
    );

    const supabase = await createServerSupabaseClient();
    const { data, error } = await supabase.rpc("cashbook", {
      p_from_date: from,
      p_to_date: to,

        {
          error: "INTERNAL_SERVER_ERROR",
          message: "Gagal mengambil data buku kas.",
        },
        { status: 500 }
      );
    }

    return NextResponse.json(data, { status: 200 });
  } catch (error) {
    console.error("Gagal memproses buku kas:", error);
    return NextResponse.json(

        error: "INTERNAL_SERVER_ERROR",

import { NextResponse } from "next/server";
import { getAdminUser } from "@/lib/auth";
import { createServerSupabaseClient } from "@/lib/supabase/server";
import {

  expenseCategoryEnum,
  getFinanceDateRange,
  financePresetEnum,
  type ExpenseCategory,

export async function GET(request: Request) {
  try {
    const admin = await getAdminUser();
    if (!admin) {
      return NextResponse.json(
        {
          error: "UNAUTHORIZED",
          message: "Akses ditolak: sesi admin diperlukan.",
        },
        { status: 401 }
      );
    }

    const { searchParams } = new URL(request.url);
    const preset = searchParams.get("preset");
    const validPreset = financePresetEnum.safeParse(preset).success
      ? (preset as "today" | "this_month" | "custom")
      : "today";
    const customFrom = searchParams.get("from") || undefined;
    const customTo = searchParams.get("to") || undefined;
    const categoryParam = searchParams.get("category");

    const { from, to } = getFinanceDateRange(validPreset, customFrom, customTo);

    const supabase = await createServerSupabaseClient();
    let query = supabase
      .from("expenses")
      .select("*")
      .gte("expense_date", from)
      .lte("expense_date", to)
      .order("expense_date", { ascending: false })
      .order("created_at", { ascending: false });

    if (categoryParam && expenseCategoryEnum.safeParse(categoryParam).success) {
      query = query.eq("category", categoryParam as ExpenseCategory);
    }

    const { data, error } = await query;
    if (error) {
      console.error("Gagal mengambil daftar pengeluaran:", error);
      return NextResponse.json(
        {
          error: "INTERNAL_SERVER_ERROR",
          message: "Gagal mengambil daftar pengeluaran.",
        },
        { status: 500 }
      );
    }

    return NextResponse.json({ expenses: data || [] }, { status: 200 });
  } catch (error) {
    console.error("Gagal memproses daftar pengeluaran:", error);
    return NextResponse.json(
      {
        error: "INTERNAL_SERVER_ERROR",
        message: "Gagal memproses data pengeluaran.",
      },
      { status: 500 }

  }

import { NextResponse } from "next/server";
import { z } from "zod";
import { getAdminUser } from "@/lib/auth";
import { createServerSupabaseClient } from "@/lib/supabase/server";
import { expensePayloadSchema } from "@/lib/admin-finance";

const uuidSchema = z.string().uuid("ID pengeluaran tidak valid.");
const updatePayloadSchema = expensePayloadSchema.partial();

export async function PATCH(
  request: Request,
  props: { params: Promise<{ id: string }> }
) {
  try {
    const admin = await getAdminUser();
    if (!admin) {
      return NextResponse.json(
        {
          error: "UNAUTHORIZED",
          message: "Akses ditolak: sesi admin diperlukan.",
        },
        { status: 401 }
      );
    }

    const { id } = await props.params;
    const parsedId = uuidSchema.safeParse(id);
    if (!parsedId.success) {
      return NextResponse.json(
        { error: "BAD_REQUEST", message: "Format ID pengeluaran tidak valid." },
        { status: 400 }
      );
    }

    const json = await request.json().catch(() => null);
    const parsed = updatePayloadSchema.safeParse(json);
    if (!parsed.success || Object.keys(parsed.data).length === 0) {
      return NextResponse.json(
        {
          error: "BAD_REQUEST",
          message: "Perubahan data pengeluaran tidak valid.",
          details: parsed.error?.issues,
        },
        { status: 400 }
      );
    }

    const supabase = await createServerSupabaseClient();
    const { data, error } = await supabase
      .from("expenses")
      .update(parsed.data)
      .eq("id", parsedId.data)
      .select()
      .single();

    if (error || !data) {
      console.error("Gagal memperbarui catatan pengeluaran:", error);
      return NextResponse.json(
        {
          error: "NOT_FOUND",
          message: "Catatan pengeluaran tidak ditemukan atau gagal diperbarui.",
        },
        { status: 404 }
      );
    }

    return NextResponse.json({ expense: data }, { status: 200 });
  } catch (error) {
    console.error("Gagal memproses perbaikan pengeluaran:", error);
    return NextResponse.json(
      {
        error: "INTERNAL_SERVER_ERROR",
        message: "Gagal memproses pembaruan pengeluaran.",

    );
  }
}

import { FinancePeriodSelector } from "./finance-period-selector";
import { FinanceSummaryCards } from "./finance-summary-cards";
import { CashbookDailyTable } from "./cashbook-daily-table";
import { ExpenseListSection } from "./expense-list-section";
import { ExpenseFormDialog } from "./expense-form-dialog";
import { DeleteExpenseDialog } from "./delete-expense-dialog";
import type { FinanceData, FinancePreset, Expense, ExpensePayload } from "@/lib/admin-finance";

interface FinanceDashboardClientProps {
  initialData: FinanceData;
  preset: FinancePreset;
  from: string;
  to: string;
  todayStr: string;
}

export function FinanceDashboardClient({
  initialData,
  preset,
  from,
  to,
  todayStr,
}: FinanceDashboardClientProps) {
  const router = useRouter();
  const [data, setData] = useState<FinanceData>(initialData);
  const [isFormOpen, setIsFormOpen] = useState(false);
  const [editingExpense, setEditingExpense] = useState<Expense | null>(null);
  const [deletingExpense, setDeletingExpense] = useState<Expense | null>(null);

  const handleCreateOrUpdateExpense = async (payload: ExpensePayload) => {
    if (editingExpense) {
      const res = await fetch(`/api/admin/expenses/${editingExpense.id}`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
      });
      if (!res.ok) {
        const body = await res.json().catch(() => ({}));
        throw new Error(body.message || "Gagal memperbarui pengeluaran.");
      }
    } else {
      const res = await fetch("/api/admin/expenses", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
      });
      if (!res.ok) {
        const body = await res.json().catch(() => ({}));
        throw new Error(body.message || "Gagal menambahkan pengeluaran.");
      }
    }

    router.refresh();
  };

  const handleDeleteExpense = async (id: string) => {
    const res = await fetch(`/api/admin/expenses/${id}`, {
      method: "DELETE",
    });
    if (!res.ok) {
      const body = await res.json().catch(() => ({}));
      throw new Error(body.message || "Gagal menghapus pengeluaran.");
    }

    router.refresh();
  };

  return (
    <div className="flex flex-col gap-6">
      {/* 1. Period Selector */}
      <FinancePeriodSelector
        currentPreset={preset}
        currentFrom={from}
        currentTo={to}
        todayStr={todayStr}
      />

      {/* 2. Three Metric Cards */}
      <FinanceSummaryCards summary={data.cashbook.summary} />

      {/* 3. Daily Cashbook Table */}
      <CashbookDailyTable daily={data.cashbook.dailyBreakdown} />

      {/* 4. Expense List Section */}
      <ExpenseListSection
        expenses={data.expenses}
        onAddExpense={() => {
          setEditingExpense(null);
          setIsFormOpen(true);
        }}
        onEditExpense={(expense) => {
          setEditingExpense(expense);
          setIsFormOpen(true);
        }}
        onDeleteExpense={(expense) => {
          setDeletingExpense(expense);
        }}
      />

      {/* Form Dialog (Create / Edit) */}
      <ExpenseFormDialog
        isOpen={isFormOpen}
        onClose={() => {
          setIsFormOpen(false);
          setEditingExpense(null);
        }}
        onSubmit={handleCreateOrUpdateExpense}
        initialData={editingExpense}
        todayStr={todayStr}
      />

      {/* Delete Confirmation Dialog */}
      <DeleteExpenseDialog
        isOpen={Boolean(deletingExpense)}
        expense={deletingExpense}
        onClose={() => setDeletingExpense(null)}
        onConfirm={handleDeleteExpense}
      />
    </div>
  );
}
```

import type { Metadata } from "next";
import { redirect } from "next/navigation";
import { connection } from "next/server";
import { Suspense } from "react";
import { getAdminUser } from "@/lib/auth";
import { createServerSupabaseClient } from "@/lib/supabase/server";
import {
  financeQuerySchema,
  getFinanceDateRange,
  getFinanceData,
  getWitaTodayDateString,
} from "@/lib/admin-finance";
import { FinanceDashboardClient } from "@/components/admin/finance/finance-dashboard-client";

export const metadata: Metadata = {
  title: "Catatan Keuangan · Dashboard Rumah Makan Sipil",
  description: "Buku kas, pencatatan pengeluaran operasional, dan laba kotor warung makan.",
};

function FinanceLoadingSkeleton() {
  return (
    <div className="flex flex-col gap-6 animate-pulse">
      <div className="h-28 rounded-xl bg-cream-100" />
      <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
        <div className="h-32 rounded-xl bg-cream-100" />
        <div className="h-32 rounded-xl bg-cream-100" />
        <div className="h-32 rounded-xl bg-cream-100" />
      </div>
      <div className="h-64 rounded-xl bg-cream-100" />
      <div className="h-64 rounded-xl bg-cream-100" />
    </div>
  );
}

interface PageProps {
  searchParams: Promise<{ [key: string]: string | string[] | undefined }>;
}

export default async function AdminFinancePage({ searchParams }: PageProps) {
  await connection();

  const admin = await getAdminUser();
  if (!admin) {
    redirect("/admin/login");
  }

  const rawParams = await searchParams;
  const normalizedParams: Record<string, string> = {};
  for (const [key, value] of Object.entries(rawParams)) {
    if (typeof value === "string" && value.trim() !== "") {
      normalizedParams[key] = value;
    }
  }

  const parsed = financeQuerySchema.safeParse(normalizedParams);
  const validatedQuery = parsed.success ? parsed.data : { preset: "today" as const };

  const todayStr = getWitaTodayDateString();
  const { from, to } = getFinanceDateRange(
    validatedQuery.preset,
    validatedQuery.from,
    validatedQuery.to
  );

  const supabase = await createServerSupabaseClient();
  const financeData = await getFinanceData(supabase, validatedQuery);

  return (
    <div className="flex flex-col gap-6">
      {/* Header Halaman */}
      <div className="border-b border-line pb-4">
        <h1 className="font-display text-2xl font-bold text-green-900">
          Catatan Keuangan
        </h1>
        <p className="text-sm text-ink-600">
          Pemasukan omzet pesanan, pengeluaran operasional warung, dan estimasi laba kotor.
        </p>
      </div>

      <Suspense fallback={<FinanceLoadingSkeleton />}>
        <FinanceDashboardClient
          initialData={financeData}
          preset={validatedQuery.preset}
          from={from}
          to={to}
          todayStr={todayStr}
        />
      </Suspense>
    </div>
  );
}

import { test, expect } from "@playwright/test";
import { createClient } from "@supabase/supabase-js";

const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL || "http://127.0.0.1:54321";
const serviceRoleKey = process.env.SUPABASE_SERVICE_ROLE_KEY;

const isLocal = supabaseUrl.includes("127.0.0.1") || supabaseUrl.includes("localhost");

test.describe("Admin Finance & Cashbook (E2E)", () => {
  test.skip(!isLocal || !serviceRoleKey, "Hanya dijalankan pada Supabase lokal dengan service role key.");

  const adminClient = createClient(supabaseUrl, serviceRoleKey || "", {
    auth: { autoRefreshToken: false, persistSession: false },
  });

  const testEmail = `admin-finance-test-${Date.now()}@warung.local`;
  const testPassword = "PasswordTest123!";
  let userId: string;

  test.beforeAll(async () => {
    // 1. Buat user admin sintetis
    const { data: userRes, error: userErr } = await adminClient.auth.admin.createUser({
      email: testEmail,
      password: testPassword,
      email_confirm: true,
    });
    if (userErr || !userRes.user) throw userErr;
    userId = userRes.user.id;

    // 2. Daftarkan ke tabel public.admins
    const { error: adminErr } = await adminClient
      .from("admins")
      .insert({ user_id: userId });
    if (adminErr) throw adminErr;
  });

  test.afterAll(async () => {
    // Teardown sintetis
    if (userId) {
      await adminClient.from("expenses").delete().neq("id", "00000000-0000-0000-0000-000000000000");
      await adminClient.from("admins").delete().eq("user_id", userId);
      await adminClient.auth.admin.deleteUser(userId);
    }
  });

  test("alur lengkap keuangan: login, cek buku kas, tambah, ubah, dan hapus pengeluaran", async ({ page }) => {
    // 1. Login admin
    await page.goto("/admin/login");
    await page.fill('input[type="email"]', testEmail);
    await page.fill('input[type="password"]', testPassword);
    await page.click('button[type="submit"]');

    await expect(page).toHaveURL(/\/admin/);

    // 2. Navigasi ke Catatan Keuangan
    await page.goto("/admin/keuangan");
    await expect(page.locator("h1")).toContainText("Catatan Keuangan");

    // 3. Verifikasi Tiga Kartu Metrik
    await expect(page.getByText("Total Pemasukan")).toBeVisible();
    await expect(page.getByText("Total Pengeluaran")).toBeVisible();
    await expect(page.getByText("Laba Kotor")).toBeVisible();

    // 4. Tambah Pengeluaran Baru
    await page.getByRole("button", { name: /tambah pengeluaran/i }).first().click();

    await expect(page.getByRole("dialog")).toBeVisible();
    await page.fill("#expense-description", "Beli Telur Ayam 2 Rak");
    await page.fill("#expense-amount", "110000");
    await page.selectOption("#expense-category", "bahan");

    await page.getByRole("button", { name: "Tambah Pengeluaran" }).click();

    // 5. Verifikasi pengeluaran muncul di daftar
    await expect(page.getByText("Beli Telur Ayam 2 Rak").first()).toBeVisible();
    await expect(page.getByText("Rp110.000").first()).toBeVisible();

    // 6. Ubah Pengeluaran
    await page.getByLabel("Ubah pengeluaran Beli Telur Ayam 2 Rak").first().click();
    await expect(page.getByRole("dialog")).toBeVisible();

    await page.fill("#expense-amount", "115000");
    await page.getByRole("button", { name: "Simpan Perubahan" }).click();

    await expect(page.getByText("Rp115.000").first()).toBeVisible();

    // 7. Hapus Pengeluaran
    await page.getByLabel("Hapus pengeluaran Beli Telur Ayam 2 Rak").first().click();
    await expect(page.getByRole("dialog")).toBeVisible();
    await expect(page.getByText("Hapus Catatan Pengeluaran?")).toBeVisible();

    await page.getByRole("button", { name: "Hapus Pengeluaran" }).click();

    // 8. Verifikasi pengeluaran terhapus dari daftar
    await expect(page.getByText("Beli Telur Ayam 2 Rak")).toHaveCount(0);
  });
});
